#include <Bounce2.h>

// Setup GPIO pin to trigger a macro with a button press
#define BUTTON_PIN 0
Bounce triggerBtn = Bounce();
bool macroRunning = false;

// NSGamepad button indices
#define BUTTON_Y_INDEX 0        // Button Y
#define BUTTON_B_INDEX 1        // Button B 
#define BUTTON_A_INDEX 2        // Button A
#define BUTTON_X_INDEX 3        // Button X
#define BUTTON_L_INDEX 4        // Left shoulder button L
#define BUTTON_R_INDEX 5        // Right shoulder button R
#define BUTTON_ZL_INDEX 6       // Left trigger ZL
#define BUTTON_ZR_INDEX 7       // Right trigger ZR
#define BUTTON_MINUS_INDEX 8    // Minus button
#define BUTTON_PLUS_INDEX 9     // Plus button
#define BUTTON_LCLICK_INDEX 10  // Left stick click
#define BUTTON_RCLICK_INDEX 11  // Right stick click
#define BUTTON_HOME_INDEX 12    // Home button
#define BUTTON_CAPTURE_INDEX 13 // Capture button

// DPad direction values 
#define DPAD_CENTERED 0xF
#define DPAD_UP 0
#define DPAD_UP_RIGHT 1
#define DPAD_RIGHT 2
#define DPAD_DOWN_RIGHT 3
#define DPAD_DOWN 4
#define DPAD_DOWN_LEFT 5
#define DPAD_LEFT 6
#define DPAD_UP_LEFT 7

void setup() {
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  triggerBtn.attach(BUTTON_PIN);
  triggerBtn.interval(10); // Debounce 10ms
  NSGamepad.begin();       // Initialize NSGamepad
}

// Helper: Press and release NSGamepad button by index
void pressButton(uint8_t index, unsigned int pressDuration = 500, unsigned int releaseDuration = 500) {
  NSGamepad.press(index);
  NSGamepad.loop();
  delay(pressDuration);
  NSGamepad.release(index);
  NSGamepad.loop();
  delay(releaseDuration);
}

// Helper: Set DPad direction then return to center
void setDPad(uint8_t direction, unsigned int duration = 500) {
  NSGamepad.dPad(direction);
  NSGamepad.loop();
  delay(duration);
  NSGamepad.dPad(DPAD_CENTERED);
  NSGamepad.loop();
  delay(duration);
}

// Helper: Hold left stick in a direction
void holdStick(int8_t x, int8_t y, unsigned int duration) {
  // Convert -128..127 to 0..255 for NSGadget
  NSGamepad.leftXAxis((uint8_t)(x + 128));
  NSGamepad.leftYAxis((uint8_t)(y + 128));
  NSGamepad.loop();
  delay(duration);
  
  // Reset to center (128 = neutral position)
  NSGamepad.leftXAxis(128);
  NSGamepad.leftYAxis(128);
  NSGamepad.loop();
  delay(200); // Short pause after resetting
}

void runMacro() {
  // Step 1: Press A three times to register controller and close pairing menu
  pressButton(BUTTON_A_INDEX, 500, 500);  // A - 1st press
  pressButton(BUTTON_A_INDEX, 500, 2000); // A - 2nd press and wait
  pressButton(BUTTON_A_INDEX, 500, 1000); // A - 3rd press 
  
  // Step 2: Wait for menu to close
  delay(1000);
  
  // Step 3: Press L twice quickly to set pen size
  pressButton(BUTTON_L_INDEX, 50, 50);  // L - quick press
  pressButton(BUTTON_L_INDEX, 50, 100); // L - quick press with slightly longer pause
  
  // Step 4: Clear the canvas
  pressButton(BUTTON_LCLICK_INDEX, 50, 50); // LStick - quick click
    
  // Step 5: Move left analog stick to top-left corner for 7 seconds
  holdStick(-128, -128, 7000); // Full top-left for 7 seconds
  
  // Step 6: Run drawing macro
  drawImage();
  
  // Step 7: Save and exit
  pressButton(BUTTON_MINUS_INDEX, 50, 50); // Minus - quick press
}

void loop() {
  triggerBtn.update();
  
  // Trigger macro when button is pressed (falling edge)
  if (!macroRunning && triggerBtn.fell()) {
    macroRunning = true;
    runMacro();
    macroRunning = false;
  }
  
  // Keep USB reports fresh
  NSGamepad.loop();
}
